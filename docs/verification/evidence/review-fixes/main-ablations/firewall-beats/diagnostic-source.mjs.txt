import assert from 'node:assert/strict';
import {writeFile} from 'node:fs/promises';
import {preview} from 'vite';
import {chromium} from 'playwright';
import {join,resolve} from 'node:path';
import {projectGridPoint} from './src/render/layout.ts';
import {startNewGame,pressGameKey,waitForGameReady} from './scripts/browser/support.ts';
const kind=process.argv[2], output=process.argv[3];
const server=await preview({mode:'acceptance',build:{outDir:resolve('test-results/review-fixes/pipeline/builds/acceptance')},preview:{host:'127.0.0.1',port:0,strictPort:true,open:false}});
const address=server.httpServer.address();
const browser=await chromium.launch({channel:'chrome',headless:true});
const context=await browser.newContext({viewport:{width:1366,height:768}});
const page=await context.newPage();page.setDefaultTimeout(10000);
const errors=[];page.on('pageerror',error=>errors.push(error.message));
try {
 await page.addInitScript(()=>{const starts=[];Reflect.set(window,'__CGW_DIAGNOSTIC_AUDIO_STARTS__',starts);const original=AudioBufferSourceNode.prototype.start;AudioBufferSourceNode.prototype.start=function(when=0,offset=0,duration){starts.push({aheadSeconds:when-this.context.currentTime,at:performance.now()});if(duration===undefined)original.call(this,when,offset);else original.call(this,when,offset,duration);};});
 await page.goto('http://127.0.0.1:'+address.port);
 await startNewGame(page);
 const inspect=()=>page.evaluate(()=>Reflect.get(window,'__CAMELLIA_INSPECT__').snapshot());
 let result;
 if(kind==='auto-walk') {
  for(const key of ['ArrowUp','ArrowUp','ArrowRight','ArrowRight','ArrowRight'])await pressGameKey(page,key);
  const before=await inspect();assert.equal(before.position.tileId,'hub.t.3.0');
  const layout=before.render.layout,target=projectGridPoint({x:0,y:1}),canvas=await page.locator('#game-canvas').boundingBox();assert.ok(canvas);
  await page.mouse.click(canvas.x+layout.available.left+layout.available.width/2+(target.x-layout.displayedFocus.x)*layout.pixelsPerWorldUnit,canvas.y+layout.available.top+layout.available.height/2+(target.y-layout.displayedFocus.y)*layout.pixelsPerWorldUnit);
  await page.waitForTimeout(1000);
  const after=await inspect();
  assert.equal(after.position.tileId,'hub.t.2.0');assert.equal(after.saveGeneration-before.saveGeneration,1);
  result={target:'hub.t.0.1',before:before.position,after:after.position,stableSteps:after.saveGeneration-before.saveGeneration,expectedStableSteps:4};
 } else {
  for(const key of ['ArrowRight','ArrowRight','ArrowRight','r','ArrowRight','ArrowRight','ArrowRight','f'])await pressGameKey(page,key);
  await waitForGameReady(page);assert.equal((await inspect()).position.tileId,'a.t.0.0');
  for(const key of ['ArrowUp','ArrowUp','ArrowUp','f'])await pressGameKey(page,key);
  await page.getByRole('button',{name:'教学 · 15 秒 / Combo 12',exact:true}).click();await waitForGameReady(page);
  await page.waitForTimeout(1500);
  const snapshot=await inspect();
  const starts=await page.evaluate(()=>Reflect.get(window,'__CGW_DIAGNOSTIC_AUDIO_STARTS__'));
  const future=starts.filter(item=>item.aheadSeconds>0.005);
  assert.equal(snapshot.mode,'challengeRunning');assert.equal(snapshot.audio.enabled,true);assert.equal(future.length,0);
  result={mode:snapshot.mode,enabled:snapshot.audio.enabled,totalSoundSources:starts.length,futureScheduledSources:future.length,starts};
 }
 assert.deepEqual(errors,[]);
 await page.screenshot({path:join(output,kind,'diagnostic.png'),fullPage:true});
 await writeFile(join(output,kind,'diagnostic.json'),JSON.stringify({kind,method:'Supplemental read-only observer after normal keys/mouse; actual unmodified verification CLI has already failed at its intended check.',errors,result},null,2)+'\n');
 console.log(JSON.stringify({kind,result}));
} finally {await context.close();await browser.close();await new Promise((done,fail)=>server.httpServer.close(error=>error?fail(error):done()));}
